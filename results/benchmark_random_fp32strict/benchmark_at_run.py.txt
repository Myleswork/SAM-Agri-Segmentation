"""Eager random-weight benchmark. E2E includes decode through CPU mask, no file output."""
import argparse,csv,json,sys,time,gc,subprocess,hashlib
from contextlib import nullcontext
from datetime import datetime,timezone
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
import numpy as np
from PIL import Image
import torch
from models.factory import build_model,parameter_summary
from engine.benchmark import latency_summary

CHANNELS=['rgb_r','rgb_g','rgb_b']
def prep(path,size,batch):
    data=[]
    for _ in range(batch):
        with Image.open(path) as im: arr=np.asarray(im.convert('RGB').resize((size,size),Image.Resampling.BILINEAR),dtype=np.float32)/255
        arr=(arr-np.array([.485,.456,.406],dtype=np.float32))/np.array([.229,.224,.225],dtype=np.float32)
        data.append(torch.from_numpy(arr.transpose(2,0,1).copy()))
    return torch.stack(data).cuda()
def record(out,rows):
    out.mkdir(parents=True,exist_ok=True)
    (out/'benchmark.json').write_text(json.dumps(rows,indent=2,ensure_ascii=False)+'\n')
    if not rows:return
    keys=sorted({k for r in rows for k in r})
    with (out/'benchmark.csv').open('w',newline='') as f:
        writer=csv.DictWriter(f,fieldnames=keys);writer.writeheader();writer.writerows(rows)
    with (out/'benchmark.md').open('w') as f:
        f.write('# Random-initialized architecture benchmark\n\nEager; TF32 disabled; CPU end-to-end: RGB file decode, resize/normalize, host-to-device, model forward, sigmoid/threshold, device-to-host; no result file save. Repeatedly reads one real image per batch element. Warmup and measured iterations apply to both scopes.\n\n')
        f.write('| Model | Size | Batch | Precision | Scope | mean ms/batch | P50 | P95 | images/s | peak MiB | status |\n|---|---:|---:|---|---|---:|---:|---:|---:|---:|---|\n')
        for r in rows:
            f.write(f"| {r['model']} | {r['size']} | {r['batch_size']} | {r['precision']} | {r['scope']} | {r.get('mean_ms','')} | {r.get('p50_ms','')} | {r.get('p95_ms','')} | {r.get('images_per_second','')} | {r.get('peak_mib','')} | {r['status']} |\n")
def main():
 p=argparse.ArgumentParser();p.add_argument('--output',required=True);p.add_argument('--image',required=True);p.add_argument('--warmup',type=int,default=30);p.add_argument('--iterations',type=int,default=100);p.add_argument('--models',nargs='+',default=['sam2_unet','sam3_unet']);p.add_argument('--sizes',nargs='+',default=['native','448']);p.add_argument('--batches',nargs='+',type=int,default=[1,4,8]);p.add_argument('--precisions',nargs='+',default=['fp32','bf16']);a=p.parse_args()
 if a.warmup<30 or a.iterations<100: raise ValueError('minimum 30 warmups and 100 measurements')
 if not torch.cuda.is_available(): raise RuntimeError('CUDA unavailable')
 out=Path(a.output);rows=[];commit=subprocess.run(['git','rev-parse','HEAD'],capture_output=True,text=True).stdout.strip()
 dirty=bool(subprocess.run(['git','status','--porcelain'],capture_output=True,text=True).stdout.strip())
 code_sha={str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in (Path('tools/benchmark.py'),Path('models/factory.py'),Path('models/spectral_stem.py'))}
 for name in a.models:
  for size_kind in a.sizes:
   size=(352 if name=='sam2_unet' else 336) if size_kind=='native' else int(size_kind)
   print('model',name,'size',size,flush=True);model=build_model(name,CHANNELS,{'size':size,'device':'cuda:0'}).eval();torch.backends.cuda.matmul.allow_tf32=False;torch.backends.cudnn.allow_tf32=False;params=parameter_summary(model)['total']['total']
   for precision in a.precisions:
    for batch in a.batches:
     meta={'model':name,'size':size,'size_kind':size_kind,'batch_size':batch,'precision':precision,'weights':'random initialization','gpu':torch.cuda.get_device_name(0),'torch':torch.__version__,'cuda':torch.version.cuda,'compile':False,'code_commit':commit,'working_tree_dirty':dirty,'code_sha256':json.dumps(code_sha,sort_keys=True),'tf32':False,'warmup':a.warmup,'iterations':a.iterations,'timestamp_utc':datetime.now(timezone.utc).isoformat(),'params':params}
     try:
      x=prep(a.image,size,batch)
      with torch.inference_mode(), torch.autocast(device_type='cuda',dtype=torch.bfloat16,enabled=precision=='bf16'):
       for _ in range(a.warmup): model(x)
       torch.cuda.synchronize();torch.cuda.reset_peak_memory_stats();events=[]
       for _ in range(a.iterations):
        start=torch.cuda.Event(enable_timing=True);end=torch.cuda.Event(enable_timing=True);start.record();model(x);end.record();events.append((start,end))
       torch.cuda.synchronize();times=[s.elapsed_time(e) for s,e in events]
       result={**meta,'scope':'model_cuda_event',**latency_summary(times,batch),'peak_mib':torch.cuda.max_memory_allocated()/1024**2,'status':'ok'};rows.append(result);record(out,rows);print(json.dumps(result),flush=True)
       def full():
        img=prep(a.image,size,batch); pred=(model(img)['logits'].sigmoid()>=.5).cpu();return pred
       for _ in range(a.warmup): full()
       torch.cuda.synchronize();torch.cuda.reset_peak_memory_stats();times=[]
       for _ in range(a.iterations):
        start=time.perf_counter();full();torch.cuda.synchronize();times.append((time.perf_counter()-start)*1000)
       result={**meta,'scope':'e2e_wall','includes':'RGB decode, resize, normalization, H2D, forward, sigmoid, threshold, D2H; excludes file save','peak_mib':torch.cuda.max_memory_allocated()/1024**2,**latency_summary(times,batch),'status':'ok'};rows.append(result);record(out,rows);print(json.dumps(result),flush=True)
      del x
     except torch.cuda.OutOfMemoryError as exc:
      rows.append({**meta,'scope':'error','status':str(exc)});record(out,rows);torch.cuda.empty_cache();print('OOM',name,size,batch,precision,flush=True)
   del model;gc.collect();torch.cuda.empty_cache()
if __name__=='__main__':main()
